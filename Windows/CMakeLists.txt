cmake_minimum_required(VERSION 3.24)
project(CiBarWindows VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(cibar-core INTERFACE)
target_include_directories(cibar-core INTERFACE src vendor)
add_executable(cibar-tests tests/core_tests.cpp)
target_link_libraries(cibar-tests PRIVATE cibar-core)
enable_testing()
add_test(NAME core COMMAND cibar-tests "${CMAKE_CURRENT_SOURCE_DIR}/../Resources" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/mac-v1.json" "${CMAKE_CURRENT_BINARY_DIR}/windows-v1-roundtrip.json")
if(WIN32)
  enable_language(RC)
  add_executable(CiBar WIN32 src/app.cpp app.rc)
  target_include_directories(CiBar PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_options(CiBar PRIVATE /MANIFEST:NO)
  target_link_libraries(CiBar PRIVATE cibar-core d3d11 d2d1 dwrite dcomp dxgi comctl32 comdlg32 shell32 ole32 wtsapi32 advapi32)
  target_compile_definitions(CiBar PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_custom_command(TARGET CiBar POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:CiBar>/Resources")
  foreach(resource words.json ATTRIBUTIONS.txt DATA-REVIEW.json HSK4-CORRECTIONS-REVIEW.json)
    add_custom_command(TARGET CiBar POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../Resources/${resource}" "$<TARGET_FILE_DIR:CiBar>/Resources/${resource}")
  endforeach()
endif()
